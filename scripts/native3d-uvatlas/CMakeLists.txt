cmake_minimum_required(VERSION 3.21)
project(Native3DUVAtlas LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED UVATLAS_SOURCE_DIR)
  message(FATAL_ERROR "UVATLAS_SOURCE_DIR must name pinned Microsoft UVAtlas source")
endif()
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(UVATLAS_USE_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory("${UVATLAS_SOURCE_DIR}" uvatlas)
add_executable(patch patch.cpp)
target_link_libraries(patch PRIVATE UVAtlas)
